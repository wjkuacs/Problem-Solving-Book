% 1-37 题 图1_32：惠更斯等时摆——上滚轮线和下滚轮线
% 轮半径 R=1，上滚轮线 P 点轨迹；下滚轮线关于 MN 翻转
\begin{tikzpicture}[scale=1.0, font=\small]
  \def\R{1.0}
  % MN 水平线
  \draw[thick, gray!70] (-0.5,\R) -- (7.5,\R) node[right] {MN};
  % 上滚轮线（圆心在 (t, R)，P 起点在 (0, 2R)→滚动后 P 轨迹）
  \draw[thick, blue!70!black] plot[domain=0:6.28, samples=80]
    ({(\x)*\R/(pi) + \R*sin(\x r)}, {\R + \R*(1-cos(\x r))}) ;
  % 下滚轮线（关于 MN 对称）
  \draw[thick, red!60!black, dashed] plot[domain=0:6.28, samples=80]
    ({(\x)*\R/(pi) + \R*sin(\x r)}, {\R - \R*(1-cos(\x r))}) ;
  % 小球在下滚轮线上
  \fill[red!60!black] ({3.14*\R/pi + \R*sin(3.14 r)}, {\R - \R*(1-cos(3.14 r))}) circle (2pt) node[below right] {小球};
  % P 标注（上滚轮线最低点=切点）
  \fill[blue!70!black] ({0}, {2*\R}) circle (1.8pt) node[above right] {$P$};
  % R 标注
  \draw[<->, >=Stealth] ({3.14*\R/pi}, {\R}) -- ({3.14*\R/pi}, {2*\R}) node[midway,right] {$R$};
\end{tikzpicture}
