\begin{tikzpicture}[>=Stealth, thick]
    % 定义原点 (圆心) 和半径
    \coordinate (O) at (0, 0);
    \def\R{2.5}
    
    % 画圆和圆心
    \draw (O) circle (\R);
    \fill (O) circle (2pt);
    
    % 画水平虚线
    \draw[dashed] (-\R, 0) -- (\R, 0);
    
    % 根据内接直角三角形的几何关系定义顶点坐标：
    % \theta 在图中目测约为 30度。
    % 左下端点 A 角度为 180+30 = 210度
    % 右上端点 B 角度为 30度
    % 底部端点 C 与 A 的圆心角距离为 60度 (弦长为R)，所以角度为 270度
    \coordinate (A) at (210:\R);
    \coordinate (B) at (30:\R);
    \coordinate (C) at (270:\R);
    
    % 绘制直径 (2R)
    \draw (A) -- (B) node[pos=0.75, above left] {$2R$};
    
    % 绘制弦 (R) 与 弦 (\sqrt{3}R)
    \draw (A) -- (C) node[midway, below left] {$R$};
    \draw (C) -- (B) node[midway, right] {$\sqrt{3}R$};
    
    % 标注左侧虚线上的 R
    \node[above] at (180:1.3) {$R$};
    
    % 绘制并标注角度 \theta
    \draw (180:0.8) arc (180:210:0.8);
    \node at (195:1.1) {$\theta$};
\end{tikzpicture}
