% Ensure required TikZ libraries are loaded for decorations
\usetikzlibrary{decorations.pathreplacing,decorations.markings}
\begin{tikzpicture}[>=Stealth, thick]
    % 定义内外圆的半径参数
    \def\R{1.8}
    \def\h{1.2}
    
    % 定义关键点坐标
    \coordinate (O) at (0, 0);
    \coordinate (A) at (\R+\h, 0);
    \coordinate (B) at (-\R, 0);
    \coordinate (C) at (0, \R);
    
    % 画内外同心圆
    \draw (O) circle (\R);
    \draw (O) circle (\R+\h);
    
    % 画水平和竖直的虚线辅助线
    \draw[dashed] (B) -- (A);
    \draw[dashed] (O) -- (C);
    
    % 标出中心点及圆上的交点位置
    \fill (O) circle (1.5pt) node[below left=1pt] {$O$};
    \fill (A) circle (1.5pt) node[right=2pt] {$A$};
    \node[left=2pt] at (B) {$B$};
    \node[below left=2pt] at (C) {$C$};
    
    % 标出 90° 角及弧线
    \draw (0.35, 0) arc (0:90:0.35);
    \node at (0.7, 0.4) {$90^\circ$};
    
    % 底部花括号标注 R 和 h (使用 decorations.pathreplacing)
    \draw[decorate, decoration={brace, amplitude=5pt, mirror}] 
        (0, -0.15) -- (\R, -0.15) node[midway, below=5pt] {$R$};
    \draw[decorate, decoration={brace, amplitude=5pt, mirror}] 
        (\R, -0.15) -- (\R+\h, -0.15) node[midway, below=5pt] {$h$};
        
    % 绘制内部较低的转移轨道 (从 C 相切出发到 A)
    \draw[decoration={markings, mark=at position 0.65 with {\arrow{<}}}, postaction={decorate}] 
        (C) to[out=0, in=120] (A);
        
    % 绘制外部较高的转移轨道 (从内圆左上方相切出发到 A)
    % 假设起点在 135° 位置，则顺时针切线方向为 45°
    \coordinate (P_start) at (135:\R);
    \draw[decoration={markings, mark=at position 0.65 with {\arrow{<}}}, postaction={decorate}] 
        (P_start) to[out=45, in=90] (A);
\end{tikzpicture}
