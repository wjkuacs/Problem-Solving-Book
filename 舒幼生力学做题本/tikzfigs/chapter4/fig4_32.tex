% 4-32 题 图4_32：飞船绕月球圆轨道，两种登月方式
\begin{tikzpicture}[font=\small]
  % 月球
  \filldraw (0,0) circle (1.2);
  \node at (0,0) {月球};
  \draw[thick] (0,0) circle (1.2);
  % 月球中心
  \filldraw (0,0) circle (1.5pt);
  % 月球半径R
  \draw[<->] (0,0) -- (1.2,0) node[midway, below] {$R$};
  % 圆轨道（半径R+h）
  \draw[thick, dashed] (0,0) circle (2.5);
  % 高度h标注
  \draw[<->] (1.2,0) -- (2.5,0) node[midway, below] {$h$};
  % A点（圆轨道上，顶部）
  \coordinate (A) at (0,2.5);
  \filldraw (A) circle (2pt) node[above] {$A$};
  % B点（月球表面左下，方式1到达点）
  \coordinate (B) at ({-1.2*cos(30)},{-1.2*sin(30)});
  \filldraw (B) circle (2pt) node[below left] {$B$};
  % C点（月球表面右侧，方式2到达点）
  \coordinate (C) at ({1.2*cos(30)},{-1.2*sin(30)});
  \filldraw (C) circle (2pt) node[below right] {$C$};
  % 方式1轨道（A到B，椭圆转移轨道，与月球表面相切于B）
  \draw[thick, blue!70!black] (A) .. controls (-1.5,1.5) and (-2.5,0.5) .. (B);
  \node[left] at (-1.8,1.2) {(1)};
  % 方式2轨道（A到C，向外侧喷气）
  \draw[thick, red!70!black] (A) .. controls (1.5,1.5) and (2.8,0.3) .. (C);
  \node[right] at (1.8,1.2) {(2)};
  % 喷气方向标注
  \draw[->, thick, blue!70!black] (A) -- (-0.5,2.5) node[above] {向前喷};
  \draw[->, thick, red!70!black] (A) -- (0,3.2) node[above] {向外侧喷};
  % 圆轨道上的速度方向
  \draw[->, thick] (2.5,0) -- (2.5,0.5) node[right] {$v_0$};
\end{tikzpicture}
