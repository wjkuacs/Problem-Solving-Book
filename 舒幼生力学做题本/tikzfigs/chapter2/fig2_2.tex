\begin{tikzpicture}[font=\small]
  % 天花板
  \draw[thick] (-2.5,4) -- (2.5,4);
  \foreach \x in {-2.5,-2,...,2.5} {
    \draw (\x,4) -- (\x-0.3,4.3);
  }

  % 悬挂点
  \draw[thick] (0,4) -- (0,3.5);
  \node[above right] at (0.1,4) {悬挂点};

  % 小圆环 P
  \draw[thick, fill=white] (0,3.5) circle (0.12);
  \node[above right] at (0.15,3.6) {$P$};

  % 竖直轴（虚线）
  \draw[dashed, gray] (0,3.5) -- (0,0);

  % 质量 m1（左侧）
  \coordinate (O) at (0,3.5);
  \coordinate (m1) at (-1.5,1.2);
  \coordinate (m2) at (1.2,2.0);

  % 绳到 m1
  \draw[thick] (O) -- (m1);
  % 绳到 m2
  \draw[thick] (O) -- (m2);

  % 小球 m1
  \draw[fill=gray!50] (m1) circle (0.2);
  \node[below] at (m1) {$m_1$};

  % 小球 m2
  \draw[fill=gray!50] (m2) circle (0.2);
  \node[below] at (m2) {$m_2$};

  % 标注距离 l1 和 l2
  \node[left] at (-0.9,2.2) {$l_1$};
  \node[right] at (0.8,2.7) {$l_2$};

  % 角度标注
  \draw[thin, <->] (-1.0,3.0) arc[start angle=180, end angle=240, radius=0.6];
  \node[left] at (-1.3,2.8) {$\theta_1$};
  \draw[thin, <->] (0.7,3.0) arc[start angle=0, end angle=-50, radius=0.6];
  \node[right] at (1.0,2.7) {$\theta_2$};

  % 旋转方向
  \draw[->, thick] (0.5,0.5) arc[start angle=0, end angle=120, radius=0.4];
  \node[below right] at (0.6,0.3) {$\omega$};
\end{tikzpicture}
