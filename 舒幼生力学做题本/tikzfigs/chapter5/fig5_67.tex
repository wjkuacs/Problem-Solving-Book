\begin{tikzpicture}[>=Stealth, thick]
  % 斜面
  \def\ang{30}
  \coordinate (O) at (0,0);
  \coordinate (E) at (7,0);
  \draw (O) -- (E);
  \draw (O) -- ($(O)!6.5!(E)+(-0.5,3.2)$);
  % 地面短线
  \foreach \x in {0.2,0.7,...,3.5} {
    \draw (\x,0) -- ++(-0.15,-0.2);
  }
  % 斜面上的角度 \theta
  \draw (1.2,0) arc (0:\ang:1.2);
  \node[below right] at (1.05,0.25) {$\theta$};
  % 球心在斜面上的位置
  \pgfmathsetmacro{\bx}{2.2}
  \pgfmathsetmacro{\by}{\bx*tan(\ang)+0.55}
  \coordinate (C) at (\bx,\by);
  \draw (C) circle (0.55);
  \fill (C) circle (1.5pt);
  \node[left=2pt] at (C) {$m$};
  \draw[->] (C) -- ++(-0.35,0) node[midway, above] {$r$};
  % 角速度 \omega_0 (逆时针，上坡方向滚动示意)
  \draw[->] ($(C)+(-0.15,0.75)$) arc (140:40:0.45);
  \node[above] at ($(C)+(0.1,0.95)$) {$\omega_0$};
\end{tikzpicture}
