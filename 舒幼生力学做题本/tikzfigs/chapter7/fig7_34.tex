% fig7_34.tex — 7-51: 横杆 + 细线 + 弹簧 + 小球A、B
\begin{tikzpicture}[font=\small,>=stealth]
  % 天花板/横杆（高1m）
  \draw[thick] (-1,3.5) -- (3,3.5);
  \foreach \x in {-0.8,-0.6,...,2.8}
    \draw (\x,3.5) -- (\x-0.1,3.7);
  \node[above] at (1,3.6) {横杆};
  % 细线（从横杆到小球A）
  \draw[thick] (1,3.5) -- (1,2.2);
  % 小球A
  \draw[thick,fill=blue!20] (1,2.2) circle (0.2);
  \node at (1,2.2) {$A$};
  % 弹簧（A到B）
  \draw[thick] (1,2.2) -- ++(0.15,0)
    foreach \i in {1,...,4}{
      -- ++(0.1,{(-1)^\i*0.18})
    } -- (1.7,1.2);
  \node[right,font=\footnotesize] at (1.8,1.7) {$k$};
  % 小球B
  \draw[thick,fill=red!20] (1.7,1.2) circle (0.2);
  \node at (1.7,1.2) {$B$};
  % 橡皮泥（地面）
  \draw[thick,fill=orange!30] (1.2,-0.3) rectangle (2.2,0);
  \node[below] at (1.7,0) {橡皮泥};
  % 地面
  \draw[thick] (-1,0) -- (3,0);
  % 高度标注
  \draw[<->] (-0.8,0) -- (-0.8,3.5) node[midway,left] {$1$ m};
  % 弹簧伸长标注
  \node[font=\footnotesize] at (1.2,1.7) {伸长 3.0 cm};
\end{tikzpicture}
