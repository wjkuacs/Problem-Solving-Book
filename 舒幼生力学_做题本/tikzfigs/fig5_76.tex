\begin{tikzpicture}[>=Stealth, thick]
    % 不规则外轮廓
    \draw plot [smooth cycle, tension=0.7] coordinates {
        (-1.5, 0) (-1, 3.5) (1.5, 4.5) (3.5, 3) (3.5, 0.5) (2, -1.5) (0, -1)
    };
    
    % 内部三个点
    \coordinate (P2) at (0, 0);
    \coordinate (P1) at (0.2, 3);
    \coordinate (P3) at (2, 0.5);
    
    % 绘制虚线连接的三角形及边长标签
    \draw[dashed] (P1) -- (P2) node[midway, left] {$2l$};
    \draw[dashed] (P2) -- (P3) node[midway, below right=-2pt] {$l$};
    \draw[dashed] (P1) -- (P3) node[midway, right] {$\sqrt{3} l$};
    
    % 绘制质点并标注
    \fill (P1) circle (2pt) node[right=2pt] {$P_1$};
    \fill (P2) circle (2pt) node[below=2pt] {$P_2$};
    \fill (P3) circle (2pt) node[right=2pt] {$P_3$};
\end{tikzpicture}
