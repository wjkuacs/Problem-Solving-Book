% 图 4-59（题 4-32）：两条转移椭圆均以 O 为焦点，分别与月面相切于 B、C。
\begin{tikzpicture}[>=Stealth,thick]
  \def\rr{1.8}\def\hh{0.6}
  \pgfmathsetmacro{\ro}{\rr+\hh}
  \pgfmathsetmacro{\aone}{(\ro+\rr)/2}
  \pgfmathsetmacro{\bone}{sqrt(\ro*\rr)}
  \pgfmathsetmacro{\ecc}{\hh/\rr}
  \pgfmathsetmacro{\atwo}{\rr/(1-\ecc)}
  \pgfmathsetmacro{\btwo}{\atwo*sqrt(1-\ecc*\ecc)}
  \coordinate (O) at (0,0);
  \coordinate (A) at (\ro,0);
  \coordinate (B) at (-\rr,0);
  \coordinate (C) at (0,\rr);
  \draw (O) circle (\rr);
  \draw[thin] (O) circle (\ro);
  \draw[dashed,thin] (B)--(A) (O)--(C);
  \draw[decoration={markings,mark=at position 0.45 with {\arrow{>}}},postaction={decorate}]
    (A) arc[start angle=0,end angle=180,x radius=\aone,y radius=\bone];
  \draw[decoration={markings,mark=at position 0.5 with {\arrow{>}}},postaction={decorate}]
    (A) arc[start angle={asin(\ecc)},end angle=90,x radius=\btwo,y radius=\atwo];
  \fill (O) circle (1.5pt) node[below left=2pt] {$O$};
  \fill (A) circle (1.5pt) node[right=3pt] {$A$};
  \fill (B) circle (1.5pt) node[left=3pt] {$B$};
  \fill (C) circle (1.5pt) node[above left=3pt] {$C$};
  \draw[thin] (0.35,0)--(0.35,0.35)--(0,0.35);
  \node at (0.85,0.55) {$90^\circ$};
  \node[below=3pt] at (0.9,0) {$R$};
  \node[below=3pt] at ({\rr+\hh/2},0) {$h$};
\end{tikzpicture}
