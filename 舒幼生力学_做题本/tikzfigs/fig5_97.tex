\begin{tikzpicture}[>=Stealth, thick]
  % 圆木撞台阶：半径 R，速度 v_0，台阶高 h，接触点 A
  \def\R{1.2}
  \def\h{0.55}
  \draw (-3.2,0) -- (0.2,0);
  \foreach \x in {-3.0,-2.6,...,0.0} {
    \draw (\x,0) -- ++(-0.12,-0.18);
  }
  \draw (0.2,0) -- (0.2,\h) -- (3.2,\h);
  \draw (3.2,\h) -- (3.2,0);
  \foreach \x in {0.4,0.8,...,3.0} {
    \draw (\x,\h) -- ++(0.12,0.18);
  }
  \coordinate (A) at (0.2,\h);
  \pgfmathsetmacro{\ox}{0.2-\R*cos(40)}
  \pgfmathsetmacro{\oy}{\h+\R*sin(40)}
  \coordinate (C) at (\ox,\oy);
  \draw (C) circle (\R);
  \fill (C) circle (1.5pt) node[left=2pt] {$O$};
  \fill (A) circle (2pt) node[below right=2pt] {$A$};
  \draw[->] (C) -- ($(C)+(1.3,0)$) node[above=1pt] {$v_0$};
  \draw[->] (C) -- (A) node[midway, below right=-2pt] {$R$};
  \draw[<->] (0.65,0) -- (0.65,\h)
    node[midway, right=1pt] {$h$};
  \node[above] at (C) {$m$};
\end{tikzpicture}
