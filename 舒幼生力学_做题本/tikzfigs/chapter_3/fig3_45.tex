% 图 3-45（题 3-10）：三线扭摆，三根辐条将小物块固定在水平圆环中心。
% 原书图 3-43 与图 3-45 的图像错置；本做题本按题意纠正，保留题干图号。
\begin{tikzpicture}[>=Stealth,thick]
  \coordinate (O) at (0,0);
  \draw (O) ellipse (2 and 0.85);
  \foreach \ang in {30,150,270} {
    \coordinate (P) at ({2*cos(\ang)},{0.85*sin(\ang)});
    \draw (O)--(P);
    \draw (P)--++(0,2.7) coordinate (S);
    \draw ($(S)+(-0.32,0)$)--($(S)+(0.32,0)$);
    \foreach \dx in {-0.26,-0.06,0.14}
      \draw[thin] ($(S)+(\dx,0)$)--++(0.14,0.17);
  }
  \fill (O) circle (2.5pt);
\end{tikzpicture}
