% 图 1-48（题 1-26）：每条示意折线均由 A 连通到 C，且不越过斜边。
\begin{tikzpicture}[>=Stealth,thick]
  \coordinate (A) at (0,4);
  \coordinate (B) at (0,0);
  \coordinate (C) at (5,0);
  \draw[double,double distance=2pt,line width=0.5pt] (A)--(B)--(C)--cycle;
  \draw[dashed,rounded corners=2pt]
    (A)--(0,3)--(1.2,3)--(1.2,2)--(2.4,2)--(2.4,1)--(3.6,1)--(3.6,0)--(C);
  \draw[dashed,rounded corners=2pt]
    (A)--(0,2.5)--(1.5,2.5)--(1.5,1)--(3,1)--(3,0)--(C);
  \draw[dashed,rounded corners=2pt]
    (A)--(0,1.7)--(2.2,1.7)--(2.2,0.5)--(4.2,0.5)--(4.2,0)--(C);
  \node[above left] at (A) {$A$};
  \node[below left] at (B) {$B$};
  \node[below right] at (C) {$C$};
\end{tikzpicture}
