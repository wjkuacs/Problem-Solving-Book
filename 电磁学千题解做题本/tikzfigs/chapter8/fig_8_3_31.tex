\begin{tikzpicture}[em diagram]
% Full toroid with rectangular cross section and axial wire
\def\ax{.68}
\def\ay{.34}
\def\bx{1.42}
\def\by{.70}
\def\h{.55}
% Outer and inner cylinder walls
\draw (-\bx,0)--(-\bx,-\h) arc[start angle=180,end angle=360,x radius=\bx,y radius=\by]--(\bx,0);
\draw (0,0) ellipse[x radius=\bx,y radius=\by];
\draw (0,0) ellipse[x radius=\ax,y radius=\ay];
\draw (-\ax,-\ay*.1) arc[start angle=185,end angle=355,x radius=\ax,y radius=\ay];
% Winding lines on top and front wall
\foreach \ang in {0,15,...,345} {
  \draw ({\ax*cos(\ang)},{\ay*sin(\ang)})--({\bx*cos(\ang)},{\by*sin(\ang)});
}
\foreach \ang in {195,210,...,345} {
  \draw ({\bx*cos(\ang)},{\by*sin(\ang)})--({\bx*cos(\ang)},{\by*sin(\ang)-\h});
}
% Central straight wire along axis
\draw (0,-1.25)--(0,-\ay);
\draw (0,0)--(0,1.45);
% Dimensions a, b, h
\draw[dim] (0,.92)--(\ax,.92) node[midway,fill=white,inner sep=1pt] {$a$};
\draw[dim] (0,1.25)--(\bx,1.25) node[midway,fill=white,inner sep=1pt] {$b$};
\draw[guide] (\ax,0)--(\ax,1.02) (\bx,0)--(\bx,1.35);
\draw[guide] (-\bx,0)--(-\bx-.42,0) (-\bx,-\h)--(-\bx-.42,-\h);
\draw[dim] (-\bx-.28,-\h)--(-\bx-.28,0) node[midway,fill=white,inner sep=1pt] {$h$};
\end{tikzpicture}%
